! ====================================================================
! Cisco Catalyst 2960 Switch Configuration Script
! Project: Privacy-Aware Encrypted Network Traffic Classification (TechCorp)
! Target: Cisco Packet Tracer SW1 (Switch 2960)
! ====================================================================

enable
configure terminal
hostname SW1-TechCorp-Core

! 1. Create VLANs and assign descriptive names
vlan 10
 name HR_Dept
 exit

vlan 20
 name Finance_Dept
 exit

vlan 30
 name IT_Dept
 exit

vlan 40
 name Guest_Network
 exit

vlan 50
 name Enterprise_Server
 exit

vlan 60
 name Security_Monitoring
 exit

! 2. Configure 802.1Q Trunk Port to R1 Router (GigabitEthernet0/1)
interface GigabitEthernet0/1
 description 802.1Q Trunk Link to R1 GigabitEthernet0/0
 switchport mode trunk
 switchport trunk allowed vlan 10,20,30,40,50,60
 no shutdown
 exit

! 3. Assign Access Switchports to Endpoints

! Port Fa0/10 -> PC-HR (VLAN 10)
interface FastEthernet0/10
 description Access port for PC-HR
 switchport mode access
 switchport access vlan 10
 spanning-tree portfast
 no shutdown
 exit

! Port Fa0/20 -> PC-FIN (VLAN 20)
interface FastEthernet0/20
 description Access port for PC-FIN
 switchport mode access
 switchport access vlan 20
 spanning-tree portfast
 no shutdown
 exit

! Port Fa0/30 or Fa0/3 -> PC-IT (VLAN 30)
interface FastEthernet0/3
 description Access port for PC-IT
 switchport mode access
 switchport access vlan 30
 spanning-tree portfast
 no shutdown
 exit

! Port Fa0/4 -> PC-GUEST (VLAN 40)
interface FastEthernet0/4
 description Access port for PC-GUEST
 switchport mode access
 switchport access vlan 40
 spanning-tree portfast
 no shutdown
 exit

! Port Fa0/5 -> Server0 (VLAN 50)
interface FastEthernet0/5
 description Access port for Server0 (Enterprise Web/DNS)
 switchport mode access
 switchport access vlan 50
 spanning-tree portfast
 no shutdown
 exit

! Port Fa0/6 -> PC-MON (VLAN 60)
interface FastEthernet0/6
 description Access port for PC-MON (Traffic Analyst Workstation)
 switchport mode access
 switchport access vlan 60
 spanning-tree portfast
 no shutdown
 exit

end
write memory
